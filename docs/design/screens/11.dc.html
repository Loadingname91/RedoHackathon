'GAME JAM KICKOFF tonight!! 6:30 in WEB. We’ve got pizza (while it lasts), stickers and a pile of board games. Come solo, leave with a team. See you there'; const n = st.n || 0; const status = st.status || 'Available'; const published = !!st.published; const all = [ { text: 'Read the post', warn: false }, { text: 'Found offerings: pizza, stickers, games', warn: false }, { text: 'Matched building: WEB (from the campus list)', warn: false }, { text: 'Time found: 6:30 PM', warn: false }, { text: 'Room missing, please add it', warn: true }, { text: 'Ready for review', warn: false } ]; const steps = all.slice(0, n).map((s) => ({ text: s.text, mark: s.warn ?
